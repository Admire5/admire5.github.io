-- 30-day readmissions, the measure Medicare tracks
WITH stays AS (
  SELECT patient_id, admit_date, discharge_date,
    LEAD(admit_date) OVER (
      PARTITION BY patient_id
      ORDER BY admit_date) AS next_admit
  FROM encounters
  WHERE encounter_class = 'inpatient'
)
SELECT ROUND(100.0 * AVG(
  CASE WHEN next_admit - discharge_date <= 30
       THEN 1 ELSE 0 END), 1) AS readmit_rate_pct
FROM stays;
